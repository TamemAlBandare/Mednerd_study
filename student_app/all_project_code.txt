import 'dart:convert';
import 'package:shared_preferences/shared_preferences.dart';

class BookmarksService {
  static const String _key = 'favorite_lectures_list';

  /// ط¬ظ„ط¨ ظƒط§ظپط© ط§ظ„ظ…ط­ط§ط¶ط±ط§طھ ط§ظ„ظ…ظپط¶ظ„ط©
  static Future<List<Map<String, dynamic>>> getFavorites() async {
    final prefs = await SharedPreferences.getInstance();
    final String? data = prefs.getString(_key);
    if (data == null) return [];
    try {
      final List decoded = jsonDecode(data);
      return List<Map<String, dynamic>>.from(decoded);
    } catch (_) {
      return [];
    }
  }

  /// ط§ظ„طھط­ظ‚ظ‚ ط¥ظ† ظƒط§ظ†طھ ط§ظ„ظ…ط­ط§ط¶ط±ط© ظ…ظپط¶ظ„ط©
  static Future<bool> isFavorite(String lectureId) async {
    final favorites = await getFavorites();
    return favorites.any((item) => item['id'] == lectureId);
  }

  /// طھط¨ط¯ظٹظ„ ط­ط§ظ„ط© ط§ظ„ظ…ظپط¶ظ„ط© (ط¥ط¶ط§ظپط© / ط­ط°ظپ)
  static Future<bool> toggleFavorite({
    required Map<String, dynamic> lecture,
    required String subjectTitle,
  }) async {
    final prefs = await SharedPreferences.getInstance();
    final favorites = await getFavorites();
    final lectureId = lecture['id'];

    final index = favorites.indexWhere((item) => item['id'] == lectureId);
    bool added;

    if (index != -1) {
      favorites.removeAt(index);
      added = false;
    } else {
      final itemToSave = Map<String, dynamic>.from(lecture);
      itemToSave['subject_title'] = subjectTitle;
      favorites.insert(0, itemToSave);
      added = true;
    }

    await prefs.setString(_key, jsonEncode(favorites));
    return added;
  }
}

class FolderClipper extends CustomClipper<Path> {
  @override
  Path getClip(Size size) {
    const double radius = 14.0;
    const double tabWidth = 55.0;
    const double tabHeight = 12.0;

    final path = Path();
    path.moveTo(0, tabHeight + radius);
    path.lineTo(0, radius);
    path.quadraticBezierTo(0, 0, radius, 0);
    path.lineTo(tabWidth - radius, 0);
    path.quadraticBezierTo(tabWidth, 0, tabWidth + 8, tabHeight);
    path.lineTo(size.width - radius, tabHeight);
    path.quadraticBezierTo(size.width, tabHeight, size.width, tabHeight + radius);
    path.lineTo(size.width, size.height - radius);
    path.quadraticBezierTo(size.width, size.height, size.width - radius, size.height);
    path.lineTo(radius, size.height);
    path.quadraticBezierTo(0, size.height, 0, size.height - radius);
    path.close();
    return path;
  }

  @override
  bool shouldReclip(covariant CustomClipper<Path> oldClipper) => false;
}

class FolderBorderPainter extends CustomPainter {
  final bool isDark;
  FolderBorderPainter({required this.isDark});

  @override
  void paint(Canvas canvas, Size size) {
    const double radius = 14.0;
    const double tabWidth = 55.0;
    const double tabHeight = 12.0;

    final paint = Paint()
      ..color = const Color(0xFFC5A059)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.4;

    final path = Path();
    path.moveTo(0, tabHeight + radius);
    path.lineTo(0, radius);
    path.quadraticBezierTo(0, 0, radius, 0);
    path.lineTo(tabWidth - radius, 0);
    path.quadraticBezierTo(tabWidth, 0, tabWidth + 8, tabHeight);
    path.lineTo(size.width - radius, tabHeight);
    path.quadraticBezierTo(size.width, tabHeight, size.width, tabHeight + radius);
    path.lineTo(size.width, size.height - radius);
    path.quadraticBezierTo(size.width, size.height, size.width - radius, size.height);
    path.lineTo(radius, size.height);
    path.quadraticBezierTo(0, size.height, 0, size.height - radius);
    path.close();

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(covariant FolderBorderPainter oldPainter) => oldPainter.isDark != isDark;
}

class SecurityService {
  /// طھظپط¹ظٹظ„ ط­ط¸ط± ط§ظ„ط³ظƒط±ظٹظ† ط´ظˆطھ ظˆطھط³ط¬ظٹظ„ ط§ظ„ط´ط§ط´ط©
  static Future<void> enableSecureScreen() async {
    if (kIsWeb) return;
    try {
      if (Platform.isAndroid) {
        await FlutterWindowManager.addFlags(FlutterWindowManager.FLAG_SECURE);
      }
    } catch (e) {
      debugPrint('Error enabling secure screen: $e');
    }
  }

  /// ط¥ظ„ط؛ط§ط، ط­ط¸ط± ط§ظ„ط³ظƒط±ظٹظ† ط´ظˆطھ (ط¹ظ†ط¯ ط§ظ„ط®ط±ظˆط¬ ظ…ظ† ط´ط§ط´ط§طھ ط§ظ„ظپظٹط¯ظٹظˆ ظˆط§ظ„ظ€ PDF)
  static Future<void> disableSecureScreen() async {
    if (kIsWeb) return;
    try {
      if (Platform.isAndroid) {
        await FlutterWindowManager.clearFlags(FlutterWindowManager.FLAG_SECURE);
      }
    } catch (e) {
      debugPrint('Error disabling secure screen: $e');
    }
  }
}

class FullScreenVideoPage extends StatefulWidget {
  final VideoPlayerController controller;
  final double currentSpeed;
  final List<double> speeds;
  final String watermarkText;
  final ValueChanged<double> onSpeedChanged;

  const FullScreenVideoPage({
    super.key,
    required this.controller,
    required this.currentSpeed,
    required this.speeds,
    required this.watermarkText,
    required this.onSpeedChanged,
  });

  @override
  State<FullScreenVideoPage> createState() => _FullScreenVideoPageState();
}

class _FullScreenVideoPageState extends State<FullScreenVideoPage> {
  bool _showControls = true;
  Timer? _hideTimer;
  late double _activeSpeed;
  double _savedSpeedBeforeLongPress = 1.0;
  bool _isLongPressing2x = false;

  bool _showForwardIndicator = false;
  bool _showRewindIndicator = false;
  Timer? _indicatorTimer;

  // ظ…ط¤ظ‚طھ ظˆظ…ظˆظ‚ط¹ ط§ظ„ط¹ظ„ط§ظ…ط© ط§ظ„ظ…ط§ط¦ظٹط© ظ„ط´ط§ط´ط© ط§ظ„ط¹ط±ط¶ ط§ظ„ظƒط§ظ…ظ„
  Timer? _watermarkTimer;
  Alignment _watermarkAlignment = Alignment.topRight;
  final Random _random = Random();

  static const Color goldAccent = Color(0xFFD8B467);

  @override
  void initState() {
    super.initState();
    _enableSecureScreen();

    _activeSpeed = widget.currentSpeed;
    SystemChrome.setEnabledSystemUIMode(SystemUiMode.immersiveSticky);
    SystemChrome.setPreferredOrientations([
      DeviceOrientation.landscapeLeft,
      DeviceOrientation.landscapeRight,
      DeviceOrientation.portraitUp,
    ]);

    widget.controller.addListener(_videoListener);
    _startHideTimer();
    _startWatermarkAnimation();
  }

  Future<void> _enableSecureScreen() async {
    if (kIsWeb) return;
    try {
      if (Platform.isAndroid) {
        await FlutterWindowManager.addFlags(FlutterWindowManager.FLAG_SECURE);
      }
    } catch (e) {
      debugPrint('Error enabling secure screen in fullscreen: $e');
    }
  }

  void _startWatermarkAnimation() {
    _watermarkTimer = Timer.periodic(const Duration(seconds: 7), (timer) {
      if (mounted) {
        setState(() {
          final double x = (_random.nextDouble() * 1.6) - 0.8;
          final double y = (_random.nextDouble() * 1.4) - 0.7;
          _watermarkAlignment = Alignment(x, y);
        });
      }
    });
  }

  void _videoListener() {
    if (mounted) setState(() {});
  }

  void _startHideTimer() {
    _hideTimer?.cancel();
    _hideTimer = Timer(const Duration(seconds: 3), () {
      if (mounted && widget.controller.value.isPlaying && !_isLongPressing2x) {
        setState(() => _showControls = false);
      }
    });
  }

  void _togglePlayPause() {
    setState(() {
      if (widget.controller.value.isPlaying) {
        widget.controller.pause();
        _showControls = true;
        _hideTimer?.cancel();
      } else {
        widget.controller.play();
        _showControls = true;
        _startHideTimer();
      }
    });
  }

  void _seekForward10() {
    final target = widget.controller.value.position + const Duration(seconds: 10);
    if (target < widget.controller.value.duration) {
      widget.controller.seekTo(target);
    } else {
      widget.controller.seekTo(widget.controller.value.duration);
    }
    _showFeedback(isForward: true);
  }

  void _seekRewind10() {
    final target = widget.controller.value.position - const Duration(seconds: 10);
    widget.controller.seekTo(target > Duration.zero ? target : Duration.zero);
    _showFeedback(isForward: false);
  }

  void _showFeedback({required bool isForward}) {
    _indicatorTimer?.cancel();
    setState(() {
      if (isForward) {
        _showForwardIndicator = true;

  static Future<void> saveProgress({
    required Map<String, dynamic> lecture,
    required List<Map<String, dynamic>> allLectures,
    required String subjectTitle,
    required int positionSeconds,
    required int totalDurationSeconds,
  }) async {
    // ط¹ط¯ظ… ط§ظ„ط­ظپط¸ ط¥ط°ط§ ظƒط§ظ†طھ ط§ظ„ط«ظˆط§ظ†ظٹ ط£ظ‚ظ„ ظ…ظ† 3 ط«ظˆط§ظ†ظچ

      'allLectures': allLectures,
      'subject_title': subjectTitle,
      'position_seconds': positionSeconds,
      'total_duration_seconds': totalDurationSeconds,
      'saved_at': DateTime.now().toIso8601String(),

    await prefs.setString(_key, jsonEncode(data));

    final raw = prefs.getString(_key);
    if (raw == null) return null;
    try {
      return jsonDecode(raw) as Map<String, dynamic>;
    } catch (_) {
      return null;
    }
  }
